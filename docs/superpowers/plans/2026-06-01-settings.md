import fp from "fastify-plugin";
import type { FastifyPluginAsync } from "fastify";
import { verifyClerkToken } from "../lib/clerk.js";
import { redis } from "../lib/redis.js";
import { auditStorage } from "../lib/audit.js";

const authPlugin: FastifyPluginAsync = async (fastify) => {
  // Establish audit context for every request (populated after auth)
  fastify.addHook("onRequest", function (request, reply, done) {
    auditStorage.run(
      { orgId: "", actorId: "", actorName: null, ipAddress: request.ip },
      done
    );
  });

  fastify.addHook("preHandler", async (request, reply) => {
    const routeConfig = request.routeOptions?.config as unknown as Record<string, unknown> | undefined;
    if (routeConfig?.["public"]) return;

    // GAP-S71: super-admin impersonation via Redis token
    const impersonateToken = request.headers["x-impersonate-token"] as string | undefined;
    if (impersonateToken) {
      const raw = await redis.get(`impersonate:${impersonateToken}`);
      if (!raw) {
        return reply.status(401).send({ error: { code: "INVALID_IMPERSONATION_TOKEN", message: "Invalid or expired impersonation token" } });
      }
      const { organizationId, issuedBy } = JSON.parse(raw) as { organizationId: string; issuedBy: string };
      request.auth = { userId: issuedBy, organizationId, role: "superAdmin", permissions: {} };
      const ctx = auditStorage.getStore();
      if (ctx) Object.assign(ctx, { orgId: organizationId, actorId: issuedBy });
      return;
    }

    let userId: string;
    try {
      ({ userId } = await verifyClerkToken(request.headers.authorization));
    } catch {
      return reply.status(401).send({
        error: { code: "UNAUTHORIZED", message: "Invalid or missing token" },
      });
    }

    const user = await fastify.prisma.user.findFirst({
      where: { id: userId, isActive: true },
      select: { role: true, organizationId: true, fullName: true },
    });

    if (!user) {
      return reply.status(403).send({
        error: { code: "FORBIDDEN", message: "User not found in organization" },
      });
    }

    const member = await fastify.prisma.organizationMember.findFirst({
      where: { userId, organizationId: user.organizationId },
      select: { permissions: true },
    });
    const permissions = (member?.permissions ?? {}) as Record<string, string>;

    request.auth = { userId, organizationId: user.organizationId, role: user.role, permissions };

    // Populate audit context now that we have user identity
    const ctx = auditStorage.getStore();
    if (ctx) {
      Object.assign(ctx, {
        orgId: user.organizationId,
        actorId: userId,
        actorName: user.fullName ?? null,
      });
    }

    setImmediate(() => {
      try {
        fastify.prisma.loginLog
          .create({
            data: {
              userId,
              orgId: user.organizationId,
              ipAddress: request.ip,
              userAgent: request.headers["user-agent"] ?? null,
            },
          })
          .catch(() => {});
      } catch {}
    });
  });
};

export default fp(authPlugin);
```

// apps/api/src/lib/prisma.ts
import { PrismaClient } from "@prisma/client";
import { PrismaPg } from "@prisma/adapter-pg";
import { auditStorage } from "./audit.js";
import type { InputJsonValue } from "@prisma/client/runtime/library";

const AUDITED_OPS = new Set(["create", "update", "delete"]);

const AUDIT_MAP: Record<string, { category: string }> = {
  Contact:       { category: "contacts" },
  Campaign:      { category: "campaigns" },
  Flow:          { category: "flows" },
  Template:      { category: "templates" },
  Deal:          { category: "deals" },
  Label:         { category: "labels" },
  Segment:       { category: "segments" },
  WebhookAction: { category: "settings" },
  RoutingRule:   { category: "settings" },
  ApiKey:        { category: "api_keys" },
  CannedResponse:{ category: "settings" },
  InfoMaterial:  { category: "media" },
  CustomField:   { category: "custom_fields" },
};

function resourceLabel(model: string, result: unknown): string {
  const r = result as Record<string, unknown> | null;
  if (!r) return model;
  const name = r["name"] ?? r["title"] ?? r["shortcut"] ?? r["email"] ?? r["id"] ?? "";
  return `${model}: ${String(name)}`;
}

function createClient(): PrismaClient {
  const adapter = new PrismaPg({ connectionString: process.env["DATABASE_URL"]! });
  return new PrismaClient({
    adapter,
    log: process.env["NODE_ENV"] === "development" ? ["warn", "error"] : ["error"],
  });
}

const globalForPrisma = global as unknown as { prisma: PrismaClient };
const base = globalForPrisma.prisma ?? createClient();
if (process.env["NODE_ENV"] !== "production") globalForPrisma.prisma = base;

export const prisma = base.$extends({
  query: {
    $allModels: {
      async $allOperations({ model, operation, query, args }: {
        model: string;
        operation: string;
        query: (args: unknown) => Promise<unknown>;
        args: unknown;
      }) {
        const result = await query(args);
        const ctx = auditStorage.getStore();
        const entry = AUDIT_MAP[model];
        if (!ctx?.orgId || !entry || !AUDITED_OPS.has(operation)) return result;

        const event = `${model.toLowerCase()}.${operation}d`;
        const res = result as Record<string, unknown> | null;

        void base.auditLog.create({
          data: {
            orgId: ctx.orgId,
            actorId: ctx.actorId || null,
            actorName: ctx.actorName,
            ipAddress: ctx.ipAddress,
            event,
            category: entry.category,
            resourceId: res?.["id"] as string | undefined,
            resourceLabel: resourceLabel(model, result),
            metadata: null as InputJsonValue,
          },
        }).catch(() => {});

        return result;
      },
    },
  },
});
```

// apps/api/src/routes/audit-logs.ts
import type { FastifyPluginAsync } from "fastify";
import { prisma } from "../lib/prisma.js";

const PAGE_SIZE = 25;

export const auditLogsRouter: FastifyPluginAsync = async (fastify) => {
  fastify.get<{
    Querystring: { page?: string; category?: string; actorId?: string; from?: string; to?: string };
  }>("/audit-logs", async (request, reply) => {
    if (request.auth.role !== "admin" && request.auth.role !== "superAdmin") {
      return reply.status(403).send({ error: { code: "FORBIDDEN", message: "Admin only" } });
    }
    const { page = "1", category, actorId, from, to } = request.query;
    const skip = (Number(page) - 1) * PAGE_SIZE;
    const where = {
      orgId: request.auth.organizationId,
      ...(category && { category }),
      ...(actorId && { actorId }),
      ...(from || to
        ? {
            createdAt: {
              ...(from && { gte: new Date(from) }),
              ...(to && { lte: new Date(to) }),
            },
          }
        : {}),
    };
    const [logs, total] = await Promise.all([
      prisma.auditLog.findMany({ where, orderBy: { createdAt: "desc" }, skip, take: PAGE_SIZE }),
      prisma.auditLog.count({ where }),
    ]);
    return { data: logs, total, page: Number(page), pageSize: PAGE_SIZE };
  });

  fastify.get<{
    Querystring: { category?: string; actorId?: string; from?: string; to?: string };
  }>("/audit-logs/export", async (request, reply) => {
    if (request.auth.role !== "admin" && request.auth.role !== "superAdmin") {
      return reply.status(403).send({ error: { code: "FORBIDDEN", message: "Admin only" } });
    }
    const { category, actorId, from, to } = request.query;
    const where = {
      orgId: request.auth.organizationId,
      ...(category && { category }),
      ...(actorId && { actorId }),
      ...(from || to
        ? { createdAt: { ...(from && { gte: new Date(from) }), ...(to && { lte: new Date(to) }) } }
        : {}),
    };
    const logs = await prisma.auditLog.findMany({ where, orderBy: { createdAt: "desc" }, take: 10000 });
    const header = "timestamp,actor,event,category,resource,ip\n";
    const rows = logs
      .map((l) =>
        [
          l.createdAt.toISOString(),
          `"${l.actorName ?? l.actorId ?? ""}"`,
          l.event,
          l.category,
          `"${l.resourceLabel ?? ""}"`,
          l.ipAddress ?? "",
        ].join(",")
      )
      .join("\n");
    void reply.header("Content-Type", "text/csv");
    void reply.header("Content-Disposition", `attachment; filename="audit-log.csv"`);
    return reply.send(header + rows);
  });
};
```

// apps/api/src/routes/api-keys.ts
import type { FastifyPluginAsync } from "fastify";
import { createHash, randomBytes } from "node:crypto";
import { prisma } from "../lib/prisma.js";

function generateKey(): { key: string; prefix: string; hash: string } {
  const raw = `sk_live_${randomBytes(24).toString("hex")}`;
  const prefix = raw.slice(0, 16);
  const hash = createHash("sha256").update(raw).digest("hex");
  return { key: raw, prefix, hash };
}

export const apiKeysRouter: FastifyPluginAsync = async (fastify) => {
  fastify.get("/api-keys", async (request) => {
    const keys = await prisma.apiKey.findMany({
      where: { orgId: request.auth.organizationId },
      orderBy: { createdAt: "desc" },
      select: { id: true, name: true, prefix: true, lastUsedAt: true, createdAt: true, revokedAt: true },
    });
    return { data: keys };
  });

  fastify.post<{ Body: { name: string } }>("/api-keys", async (request, reply) => {
    if (request.auth.role !== "admin") {
      return reply.status(403).send({ error: { code: "FORBIDDEN", message: "Admin only" } });
    }
    const { key, prefix, hash } = generateKey();
    const created = await prisma.apiKey.create({
      data: { orgId: request.auth.organizationId, name: request.body.name, keyHash: hash, prefix },
    });
    return reply.status(201).send({
      data: { id: created.id, name: created.name, prefix: created.prefix, key, createdAt: created.createdAt },
    });
  });

  fastify.delete<{ Params: { id: string } }>("/api-keys/:id", async (request, reply) => {
    if (request.auth.role !== "admin") {
      return reply.status(403).send({ error: { code: "FORBIDDEN", message: "Admin only" } });
    }
    const existing = await prisma.apiKey.findFirst({
      where: { id: request.params.id, orgId: request.auth.organizationId },
    });
    if (!existing) return reply.status(404).send({ error: { code: "NOT_FOUND", message: "Key not found" } });
    const updated = await prisma.apiKey.update({
      where: { id: request.params.id },
      data: { revokedAt: new Date() },
      select: { id: true, name: true, revokedAt: true },
    });
    return { data: updated };
  });
};
```

export function SlideOver({ open, onClose, title, children }: Props): JSX.Element {
  useEffect(() => {
    const handler = (e: KeyboardEvent) => { if (e.key === "Escape") onClose(); };

  }, [open, onClose]);

  return (
    <div className="fixed inset-0 z-50 flex justify-end">
      <div className="absolute inset-0 bg-black/30" onClick={onClose} />
      <div className="relative w-full max-w-md bg-white shadow-xl flex flex-col h-full">

            onClick={onClose}
            className="text-gray-400 hover:text-gray-600 text-xl leading-none"
          >
            ×
          </button>
        </div>

// apps/web/app/(dashboard)/settings/api-keys/page.tsx
"use client";
import type { JSX } from "react";
import { useState } from "react";
import { useQuery, useMutation, useQueryClient } from "@tanstack/react-query";
import { useAuth } from "@clerk/nextjs";
import { Toast, useToast } from "@/components/ui/Toast";

const API = process.env["NEXT_PUBLIC_API_URL"] ?? "http://localhost:4000";

interface ApiKey {
  id: string; name: string; prefix: string;
  lastUsedAt: string | null; createdAt: string; revokedAt: string | null;
}

export default function ApiKeysPage(): JSX.Element {
  const { getToken } = useAuth();
  const qc = useQueryClient();
  const { toast, toastState, setToastOpen } = useToast();
  const [newName, setNewName] = useState("");
  const [showCreate, setShowCreate] = useState(false);
  const [revealedKey, setRevealedKey] = useState<{ id: string; key: string } | null>(null);
  const [copied, setCopied] = useState(false);

  async function authed(url: string, opts: RequestInit = {}) {
    const token = await getToken();
    return fetch(`${API}${url}`, { ...opts, headers: { Authorization: `Bearer ${token ?? ""}`, ...opts.headers } });
  }

  const { data } = useQuery<{ data: ApiKey[] }>({
    queryKey: ["api-keys"],
    queryFn: () => authed("/v1/api-keys").then((r) => r.json()),
  });

  const create = useMutation({
    mutationFn: () => authed("/v1/api-keys", {
      method: "POST", headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ name: newName }),
    }).then((r) => r.json() as Promise<{ data: { id: string; key: string } }>),
    onSuccess: (res) => {
      qc.invalidateQueries({ queryKey: ["api-keys"] });
      setRevealedKey({ id: res.data.id, key: res.data.key });
      setNewName(""); setShowCreate(false);
    },
  });

  const revoke = useMutation({
    mutationFn: (id: string) => authed(`/v1/api-keys/${id}`, { method: "DELETE" }).then((r) => r.json()),
    onSuccess: () => { qc.invalidateQueries({ queryKey: ["api-keys"] }); toast("Key revoked", { variant: "success" }); },
  });

  async function copy(text: string) {
    await navigator.clipboard.writeText(text);
    setCopied(true);
    setTimeout(() => setCopied(false), 2000);
  }

  const keys = data?.data ?? [];

  return (
    <div className="space-y-6">
      <div className="flex items-center justify-between">
        <div>
          <h1 className="text-2xl font-semibold text-gray-900">API Keys</h1>
          <p className="text-sm text-gray-500 mt-1">Manage programmatic access to the TrustCRM API.</p>
        </div>
        <button onClick={() => setShowCreate((v) => !v)}
          className="px-4 py-2 bg-green-600 text-white text-sm font-medium rounded-lg hover:bg-green-700">
          New Key
        </button>
      </div>

      {showCreate && (
        <div className="bg-white rounded-xl border border-gray-200 p-5 space-y-3">
          <h2 className="text-sm font-semibold text-gray-700">Create API Key</h2>
          <div className="flex gap-3">
            <input value={newName} onChange={(e) => setNewName(e.target.value)} placeholder="e.g. Production integration"
              className="flex-1 border border-gray-300 rounded-lg px-3 py-2 text-sm" />
            <button onClick={() => create.mutate()} disabled={!newName || create.isPending}
              className="px-4 py-2 bg-green-600 text-white text-sm rounded-lg disabled:opacity-50">
              {create.isPending ? "Creating…" : "Create"}
            </button>
          </div>
        </div>
      )}

      {revealedKey && (
        <div className="bg-amber-50 border border-amber-200 rounded-xl p-5 space-y-3">
          <p className="text-sm font-semibold text-amber-800">Copy your key now — it won&apos;t be shown again.</p>
          <div className="flex gap-2 items-center">
            <code className="flex-1 bg-white border border-amber-300 rounded px-3 py-2 text-xs font-mono break-all">
              {revealedKey.key}
            </code>
            <button onClick={() => copy(revealedKey.key)}
              className="px-3 py-2 bg-amber-600 text-white text-xs rounded hover:bg-amber-700">
              {copied ? "Copied!" : "Copy"}
            </button>
          </div>
          <button onClick={() => setRevealedKey(null)} className="text-xs text-amber-600 hover:underline">
            I&apos;ve saved it — dismiss
          </button>
        </div>
      )}

      <div className="bg-white rounded-xl border border-gray-200 divide-y divide-gray-100">
        {keys.length === 0 && <p className="px-5 py-8 text-center text-sm text-gray-400">No API keys yet.</p>}
        {keys.map((k) => (
          <div key={k.id} className="flex items-center justify-between px-5 py-3">
            <div>
              <div className="flex items-center gap-2">
                <p className="text-sm font-medium text-gray-900">{k.name}</p>
                {k.revokedAt && <span className="text-xs bg-red-100 text-red-700 px-2 py-0.5 rounded-full">Revoked</span>}
              </div>
              <p className="text-xs text-gray-500 font-mono">{k.prefix}…</p>
              <p className="text-xs text-gray-400">
                Created {new Date(k.createdAt).toLocaleDateString()} ·{" "}
                {k.lastUsedAt ? `Last used ${new Date(k.lastUsedAt).toLocaleDateString()}` : "Never used"}
              </p>
            </div>
            {!k.revokedAt && (
              <button onClick={() => { if (confirm("Revoke this key? It will stop working immediately.")) revoke.mutate(k.id); }}
                className="text-xs text-red-500 hover:text-red-700 border border-red-200 rounded px-3 py-1">
                Revoke
              </button>
            )}
          </div>
        ))}
      </div>
      <Toast title={toastState.title} variant={toastState.variant} open={toastState.open} onOpenChange={setToastOpen} />
    </div>
  );
}

// apps/web/app/(dashboard)/settings/audit-log/page.tsx
"use client";
import type { JSX } from "react";
import { useState } from "react";
import { useQuery } from "@tanstack/react-query";
import { useAuth } from "@clerk/nextjs";

const API = process.env["NEXT_PUBLIC_API_URL"] ?? "http://localhost:4000";

interface AuditLog {
  id: string; actorName: string | null; actorId: string | null;
  event: string; category: string; resourceLabel: string | null;
  ipAddress: string | null; createdAt: string;
}

const CATEGORY_COLORS: Record<string, string> = {
  contacts: "bg-blue-100 text-blue-700",
  campaigns: "bg-purple-100 text-purple-700",
  team: "bg-orange-100 text-orange-700",
  settings: "bg-gray-100 text-gray-700",
  auth: "bg-green-100 text-green-700",
  billing: "bg-yellow-100 text-yellow-700",
  api_keys: "bg-red-100 text-red-700",
  flows: "bg-indigo-100 text-indigo-700",
  media: "bg-pink-100 text-pink-700",
  custom_fields: "bg-teal-100 text-teal-700",
};

export default function AuditLogPage(): JSX.Element {
  const { getToken } = useAuth();
  const [page, setPage] = useState(1);
  const [category, setCategory] = useState("");
  const [from, setFrom] = useState("");
  const [to, setTo] = useState("");

  async function authed(url: string) {
    const token = await getToken();
    return fetch(`${API}${url}`, { headers: { Authorization: `Bearer ${token ?? ""}` } });
  }

  const params = new URLSearchParams({ page: String(page), ...(category && { category }), ...(from && { from }), ...(to && { to }) });

  const { data, isLoading } = useQuery<{ data: AuditLog[]; total: number; pageSize: number }>({
    queryKey: ["audit-logs", page, category, from, to],
    queryFn: () => authed(`/v1/audit-logs?${params.toString()}`).then((r) => r.json()),
  });

  async function exportCsv() {
    const exportParams = new URLSearchParams({ ...(category && { category }), ...(from && { from }), ...(to && { to }) });
    const token = await getToken();
    const res = await fetch(`${API}/v1/audit-logs/export?${exportParams.toString()}`, {
      headers: { Authorization: `Bearer ${token ?? ""}` },
    });
    const blob = await res.blob();
    const url = URL.createObjectURL(blob);
    const a = document.createElement("a");
    a.href = url; a.download = "audit-log.csv"; a.click();
    URL.revokeObjectURL(url);
  }

  const logs = data?.data ?? [];
  const total = data?.total ?? 0;
  const pageSize = data?.pageSize ?? 25;
  const totalPages = Math.ceil(total / pageSize);

  return (
    <div className="space-y-6">
      <div className="flex items-center justify-between">
        <div>
          <h1 className="text-2xl font-semibold text-gray-900">Audit Log</h1>
          <p className="text-sm text-gray-500 mt-1">Complete record of every action taken in your organization.</p>
        </div>
        <button onClick={() => void exportCsv()}
          className="px-4 py-2 border border-gray-300 text-sm rounded-lg hover:bg-gray-50">
          Export CSV
        </button>
      </div>

      <div className="flex gap-3 flex-wrap">
        <select value={category} onChange={(e) => { setCategory(e.target.value); setPage(1); }}
          className="border border-gray-300 rounded-lg px-3 py-2 text-sm">
          <option value="">All categories</option>
          {Object.keys(CATEGORY_COLORS).map((c) => (
            <option key={c} value={c} className="capitalize">{c.replace("_", " ")}</option>
          ))}
        </select>
        <input type="date" value={from} onChange={(e) => { setFrom(e.target.value); setPage(1); }}
          className="border border-gray-300 rounded-lg px-3 py-2 text-sm" placeholder="From" />
        <input type="date" value={to} onChange={(e) => { setTo(e.target.value); setPage(1); }}
          className="border border-gray-300 rounded-lg px-3 py-2 text-sm" placeholder="To" />
        {(category || from || to) && (
          <button onClick={() => { setCategory(""); setFrom(""); setTo(""); setPage(1); }}
            className="text-sm text-gray-500 hover:text-gray-700 underline">
            Clear filters
          </button>
        )}
      </div>

      <div className="bg-white rounded-xl border border-gray-200 overflow-hidden">
        {isLoading ? (
          <p className="py-8 text-center text-sm text-gray-400">Loading…</p>
        ) : logs.length === 0 ? (
          <p className="py-8 text-center text-sm text-gray-400">No audit events found.</p>
        ) : (
          <table className="w-full text-sm">
            <thead className="bg-gray-50 border-b border-gray-200">
              <tr>
                <th className="px-4 py-3 text-left text-xs font-semibold text-gray-500 uppercase">Time</th>
                <th className="px-4 py-3 text-left text-xs font-semibold text-gray-500 uppercase">Actor</th>
                <th className="px-4 py-3 text-left text-xs font-semibold text-gray-500 uppercase">Event</th>
                <th className="px-4 py-3 text-left text-xs font-semibold text-gray-500 uppercase">Resource</th>
                <th className="px-4 py-3 text-left text-xs font-semibold text-gray-500 uppercase">IP</th>
              </tr>
            </thead>
            <tbody className="divide-y divide-gray-100">
              {logs.map((log) => (
                <tr key={log.id} className="hover:bg-gray-50">
                  <td className="px-4 py-3 text-xs text-gray-500 whitespace-nowrap">
                    {new Date(log.createdAt).toLocaleString()}
                  </td>
                  <td className="px-4 py-3 text-xs text-gray-900">{log.actorName ?? log.actorId ?? "System"}</td>
                  <td className="px-4 py-3">
                    <span className={`text-xs px-2 py-0.5 rounded-full font-medium ${CATEGORY_COLORS[log.category] ?? "bg-gray-100 text-gray-700"}`}>
                      {log.event}
                    </span>
                  </td>
                  <td className="px-4 py-3 text-xs text-gray-600">{log.resourceLabel ?? "—"}</td>
                  <td className="px-4 py-3 text-xs text-gray-400 font-mono">{log.ipAddress ?? "—"}</td>
                </tr>
              ))}
            </tbody>
          </table>
        )}
      </div>

      {totalPages > 1 && (
        <div className="flex items-center justify-between text-sm text-gray-500">
          <span>{total} total events</span>
          <div className="flex gap-2">
            <button onClick={() => setPage((p) => Math.max(1, p - 1))} disabled={page === 1}
              className="px-3 py-1 border rounded hover:bg-gray-50 disabled:opacity-40">Previous</button>
            <span className="px-3 py-1">Page {page} of {totalPages}</span>
            <button onClick={() => setPage((p) => Math.min(totalPages, p + 1))} disabled={page === totalPages}
              className="px-3 py-1 border rounded hover:bg-gray-50 disabled:opacity-40">Next</button>
          </div>
        </div>
      )}
    </div>
  );
}
